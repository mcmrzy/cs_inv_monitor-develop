1. 概述

🎯

1.1 项目目标

实现光伏逆变器监控系统的本地模式/离网模式，使用户无需连接后端云服务即可管理和监控设备。该模式适用于以下场景：

  • 无网络环境：偏远地区、地下室等无互联网覆盖的场所
  • 设备现场调试：工程师现场维护、首次配置设备
  • 网络故障：云端服务不可用时的应急方案
  • 数据隐私：敏感场景下不希望数据上传云端
👤

2.2 认证绕过策略

通过 AuthGuard 白名单机制实现本地模式免登录：

class AuthGuard {
  // 公开路由 - 无需认证
  static final List<String> _publicRoutes = [
    '/splash',
    '/onboarding',
    '/login',
    '/register',
    '/forgot-password',
    '/local-mode',  // 新增：本地模式入口
  ];
  
  // 本地模式路由 - 标记后可免登录访问
  static final List<String> _localModeRoutes = [
    '/home',
    '/devices',
    '/device/',           // 设备详情
    '/local-mode',
    '/local-device/',     // 本地设备详情
    '/local-settings',
  ];
  
  static Future<String?> redirect(
    BuildContext context, 
    GoRouterState state,
  ) async {
    final currentPath = state.matchedLocation;
    
    // 公开路由直接放行
    if (_publicRoutes.contains(currentPath)) {
      return null;
    }
    
    // 检查是否处于本地模式
    final storageService = getIt<StorageService>();
    final isLocalMode = await storageService.getIsLocalMode();
    
    if (isLocalMode && _localModeRoutes.any(
      (route) => currentPath.startsWith(route)
    )) {
      return null; // 本地模式下放行
    }
    
    // 其他路由需要认证
    final token = await storageService.getToken();
    if (token == null || token.isEmpty) {
      return '/login';
    }
    
    return null;
  }
}

4.3 BLE模式（新增）

通过蓝牙低功耗协议连接设备，适用于无WiFi环境：

新增需求：项目已集成 flutter_blue_ultra 依赖，需实现BLE通信服务。 
// 新增 - BLE通信服务
class BleCommunicationService {
  // BLE服务UUID
  static const String SERVICE_UUID = "0000fff0-0000-1000-8000-00805f9b34fb";
  static const String CHAR_READ_UUID = "0000fff1-0000-1000-8000-00805f9b34fb";
  static const String CHAR_WRITE_UUID = "0000fff2-0000-1000-8000-00805f9b34fb";
  static const String CHAR_NOTIFY_UUID = "0000fff3-0000-1000-8000-00805f9b34fb";
  
  FlutterBlueUltra _flutterBlue = FlutterBlueUltra.instance;
  BluetoothDevice? _connectedDevice;
  BluetoothCharacteristic? _readChar;
  BluetoothCharacteristic? _writeChar;
  
  // 扫描BLE设备
  Future<List<BleDevice>> scanDevices({
    Duration timeout = const Duration(seconds: 10),
  }) async {
    final devices = <BleDevice>[];
    
    await _flutterBlue.startScan(
      timeout: timeout,
      withServices: [Guid(SERVICE_UUID)], // 过滤特定服务
    );
    
    _flutterBlue.scanResults.listen((results) {
      for (final result in results) {
        if (result.device.name.startsWith('CS-INV') ||
            result.device.name.startsWith('INV-')) {
          devices.add(BleDevice(
            id: result.device.id.id,
            name: result.device.name,
            rssi: result.rssi,
          ));
        }
      }
    });
    
    await _flutterBlue.stopScan();
    return devices;
  }
  
  // 连接BLE设备
  Future<bool> connect(String deviceId) async {
    try {
      final device = await _flutterBlue.connect(deviceId);
      _connectedDevice = device;
      
      // 发现服务
      final services = await device.discoverServices();
      final service = services.firstWhere(
        (s) => s.uuid.toString() == SERVICE_UUID,
      );
      
      // 获取特征值
      _readChar = service.characteristics.firstWhere(
        (c) => c.uuid.toString() == CHAR_READ_UUID,
      );
      _writeChar = service.characteristics.firstWhere(
        (c) => c.uuid.toString() == CHAR_WRITE_UUID,
      );
      
      return true;
    } catch (e) {
      return false;
    }
  }
  
  // 读取设备数据
  Future<Map<String, dynamic>> readDeviceData() async {
    if (_readChar == null) throw Exception('Not connected');
    
    final bytes = await _readChar!.read();
    return _parseDeviceData(bytes);
  }
  
  // 写入控制命令
  Future<bool> writeCommand(int cmd, List<int> params) async {
    if (_writeChar == null) throw Exception('Not connected');
    
    final bytes = [cmd, ...params];
    await _writeChar!.write(bytes);
    return true;
  }
  
  // 解析设备数据
  Map<String, dynamic> _parseDeviceData(List<int> bytes) {
    // 解析协议数据
    return {
      'power': bytes[0] | (bytes[1] << 8),      // 当前功率 (W)
      'voltage': bytes[2] | (bytes[3] << 8),    // 直流电压 (V)
      'current': bytes[4] | (bytes[5] << 8),    // 直流电流 (A)
      'temp': bytes[6],                           // 温度 (°C)
      'status': bytes[7],                         // 设备状态
    };
  }
}

// BLE设备模型
class BleDevice {
  final String id;
  final String name;
  final int rssi;
  
  BleDevice({
    required this.id,
    required this.name,
    required this.rssi,
  });
}

离线→在线同步（Reconciliation）

// 以SN为键进行数据合并
Future<void> syncLocalDataToCloud() async {
  final db = getIt<LocalDatabase>();
  final api = getIt<ApiService>();
  
  // 1. 获取待同步数据
  final pendingDevices = await db.getPendingSyncDevices();
  final pendingData = await db.getPendingSyncTelemetry();
  
  // 2. 逐个设备同步
  for (final device in pendingDevices) {
    try {
      // 查询云端是否存在
      final cloudDevice = await api.getDeviceBySN(device.sn);
      
      if (cloudDevice == null) {
        // 云端不存在，创建新设备
        await api.createDevice(device);
      } else {
        // 云端存在，合并数据（云端优先）
        final merged = _mergeDevice(cloudDevice, device);
        await api.updateDevice(merged);
      }
      
      // 标记已同步
      await db.markDeviceSynced(device.sn);
    } catch (e) {
      print('Sync failed for ${device.sn}: $e');
    }
  }
  
  // 3. 同步采集数据
  for (final data in pendingData) {
    try {
      await api.uploadTelemetry(data);
      await db.markTelemetrySynced(data.id);
    } catch (e) {
      print('Telemetry sync failed: $e');
    }
  }
}

// 合并策略：云端优先，本地修改标记pending
LocalDevice _mergeDevice(CloudDevice cloud, LocalDevice local) {
  return LocalDevice(
    id: cloud.id,
    sn: cloud.sn,
    name: cloud.name ?? local.name,
    model: cloud.model ?? local.model,
    // 本地新增字段标记为pending
    isPendingSync: local.hasUnsyncedChanges ? 1 : 0,
  );
}

5.3 冲突解决策略

场景 策略 说明
设备信息冲突 云端优先 以云端数据为准，本地修改标记为pending待审核
采集数据冲突 时间戳合并 相同时间戳的数据取云端值，不同时间戳全部保留
配置参数冲突 最新优先 以最后修改时间较新的为准
设备删除冲突 云端优先 云端已删除的设备，本地数据归档处理

6.3 BLE带宽约束

注意：BLE理论最大带宽约100KB/s，实际可用约20-50KB/s，需优化数据传输策略。
  • 数据压缩：使用二进制协议而非JSON，减少传输字节数
  • 增量更新：只传输变化的数据字段
  • 批量传输：合并多个小包为单次传输
  • 优先级队列：关键数据优先传输（告警、状态变化）
  • MTU协商：连接时协商最大传输单元（建议247字节）

7. 安全考量

7.1 本地模式安全原则

  • 凭据隔离：本地模式不存储或传输云端JWT Token
  • 最小权限：本地模式只访问设备基础功能，敏感操作需登录
  • 数据隔离：本地缓存数据与云端数据逻辑分离
  • 会话管理：本地模式会话有时间限制，超时需重新连接
// 安全隔离实现
class SecurityService {
  // 检查操作权限
  bool canPerformAction(String action, {bool isLocalMode}) {
    if (!isLocalMode) return true; // 在线模式全功能
    
    // 本地模式限制
    const allowedInLocal = [
      'view_realtime_data',
      'view_device_info',
      'basic_control',
      'wifi_config',
    ];
    
    return allowedInLocal.contains(action);
  }
  
  // 敏感操作需登录
  Future<bool> requireLoginForAction(String action) async {
    const sensitiveActions = [
      'change_password',
      'factory_reset',
      'firmware_upgrade',
      'delete_device',
    ];
    
    if (sensitiveActions.contains(action)) {
      final isAuth = await AuthGuard.isAuthenticated();
      if (!isAuth) {
        throw AuthRequiredException('此操作需要登录');
      }
      return true;
    }
    return false;
  }
}

7.3 本地通信加密

// BLE通信加密
class BleEncryption {
  // AES-128加密
  static List<int> encrypt(List<int> data, List<int> key) {
    final cipher = AES(Key(key), mode: AESMode.cbc);
    final iv = SecureRandom(16).bytes;
    final encrypted = cipher.encrypt(data, iv: iv);
    return [...iv, ...encrypted];
  }
  
  // 解密
  static List<int> decrypt(List<int> data, List<int> key) {
    final iv = data.sublist(0, 16);
    final encrypted = data.sublist(16);
    final cipher = AES(Key(key), mode: AESMode.cbc);
    return cipher.decrypt(encrypted, iv: iv);
  }
  
  // 密钥派生
  static List<int> deriveKey(String deviceId, String salt) {
    final bytes = utf8.encode(deviceId + salt);
    return sha256.convert(bytes).bytes.sublist(0, 16);
  }
}

C. API接口设计

// 本地设备HTTP API（设备内置）
// 基础地址: http://192.168.4.1 (AP模式)
// 或: http://{device_ip} (局域网模式)

// 获取设备信息
GET /api/v1/device/info
Response: {
  "sn": "CS2026001",
  "model": "CS-INV-5KW",
  "firmware": "1.2.3",
  "mac": "AA:BB:CC:DD:EE:FF"
}

// 获取实时数据
GET /api/v1/telemetry/realtime
Response: {
  "power": 3200,
  "voltage": 580,
  "current": 5.5,
  "temperature": 45,
  "daily_energy": 12.5,
  "status": 1
}

// 获取历史数据
GET /api/v1/telemetry/history?start={timestamp}&end={timestamp}
Response: {
  "data": [
    {"timestamp": 1693000000, "power": 3100, "energy": 1.2},
    ...
  ]
}

// 设备控制
POST /api/v1/device/control
Body: {
  "command": "restart",
  "params": {}
}

// WiFi配置
POST /api/v1/wifi/config
Body: {
  "ssid": "HomeWiFi",

}
